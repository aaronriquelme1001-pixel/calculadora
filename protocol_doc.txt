let crc = 0;
for (const byte of bytes) {
crc ^= byte;
for (let bit = 0; bit < 8; bit++) {
crc = (crc & 1) ? (crc >>> 1) ^ 0x8C : crc >>> 1;
}
}
return crc;
}      Kotlin   Copy  复制     fun crc8(bytes: ByteArray): Int {

}
}

}      Swift   Copy  复制     func crc8(_ bytes: [UInt8]) -> UInt8 {
